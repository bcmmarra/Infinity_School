CREATE DATABASE hospital;

-- Utilizando o banco de dados hospital.
USE hospital;

-- Pacientes Tabela e Dados
CREATE TABLE pacientes (
	id INT PRIMARY KEY AUTO_INCREMENT, -- Toda coluna deve ter uma chave primaria.
	nome VARCHAR(255) NOT NULL,
	cpf VARCHAR(11) NOT NULL UNIQUE,
	data_nascimento DATE NOT NULL,
	ativo BOOL NOT NULL DEFAULT TRUE
);

# 1 - Crie uma tabela chamada “medico” ou “medicos” que deve armazenar os dados: nome, data_nascimento, ativo, crm
CREATE TABLE medicos (
	id INT PRIMARY KEY AUTO_INCREMENT,
    nome VARCHAR(255),
    data_nascimento DATE NOT NULL,
    ativo BOOL NOT NULL DEFAULT TRUE,
    crm VARCHAR(6) NOT NULL UNIQUE
